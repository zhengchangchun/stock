"""P70 只读探测：seed21 宇宙下 topn 臂与 all-eligible 臂的持有集合宽度与区分度。
只读（file:...?mode=ro）；不写库、不落缓存。"""
import json, sqlite3, sys, time
sys.path.insert(0, "/Users/zhengchangchun/Documents/Workspace/stock")
from stocklab.candidate import replay
from stocklab.research import xsec

DB = "file:/Users/zhengchangchun/Documents/Workspace/stock/data/stocklab.db?mode=ro"
POOL, START, END = "short", "2015-01-01", "2026-09-24"

conn = sqlite3.connect(DB, uri=True); conn.row_factory = sqlite3.Row

marks = replay.rebalance_marks(conn, pool=POOL, start=START, end=END)
print(f"marks={len(marks)} first={marks[0]} last={marks[-1]} ({time.monotonic()-t0:.1f}s)", flush=True)

topn_holds, elig_holds = xsec._scan_holds(conn, marks=marks, pool=POOL)

elig_n, inter_n, jac = [], [], []
for d in marks:
    t, e = set(topn_holds.get(d, ())), set(elig_holds.get(d, ()))
    elig_n.append(len(e)); inter_n.append(len(t & e))
    jac.append(len(t & e) / len(t | e) if (t | e) else 1.0)

    "jaccard": {"min": round(min(jac), 4), "p50": round(pct(jac, .5), 4),
